#     target/linux/{generic,<target>}/config-* 写 CONFIG_DEBUG_INFO_BTF=y 会被
#     include/kernel-defaults.mk 追加的 CONFIG_KERNEL_* 行覆盖，最终仍是 not set。

#     !KERNEL_DEBUG_INFO_REDUCED —— 不显式关掉，BTF 永远选不上。
